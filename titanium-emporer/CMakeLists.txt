cmake_minimum_required(VERSION 3.20)
project(TitaniumEmporer VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(titanium-tests tests/runtime_test.cpp)
target_include_directories(titanium-tests PRIVATE include)
enable_testing()
add_test(NAME runtime COMMAND titanium-tests)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Titanium Emporer requires x64")
  endif()
  add_library(titanium-client SHARED src/client.cpp)
  set_target_properties(titanium-client PROPERTIES PREFIX "")
  add_executable(Titanium-Emporer WIN32 src/launcher.cpp src/game.cpp)
  target_include_directories(Titanium-Emporer PRIVATE include)
  target_compile_definitions(Titanium-Emporer PRIVATE UNICODE _UNICODE)
  target_link_libraries(Titanium-Emporer PRIVATE user32 gdi32 version dwmapi)
  if(MINGW)
    target_link_options(Titanium-Emporer PRIVATE -municode -static-libgcc -static-libstdc++)
    target_link_options(titanium-client PRIVATE -static-libgcc -static-libstdc++)
  endif()
endif()
